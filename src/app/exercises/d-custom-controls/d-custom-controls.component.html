💡 FormValueControl<T> interface 
// The interface requires exactly one property:
interface FormValueControl<T> {
  value: ModelSignal<T>;       // required
  disabled?: InputSignal<boolean>; // optional
}

// ❌ Before — standalone signal (not compatible with [formField])
export class StarRating {
  protected readonly currentValue = signal(0);
}

// ✅ After — satisfies FormValueControl<number>
export class StarRating implements FormValueControl<number> {
  value = model<number>(0);   // model() creates a ModelSignal
}

// Now [formField] just works:
<app-star-rating [formField]="form.rating" />
💡 ControlValueAccessor vs FormValueControl 
// ❌ OLD — ControlValueAccessor (~50 lines of boilerplate)
@Component({
  providers: [{
    provide: NG_VALUE_ACCESSOR,
    useExisting: forwardRef(() => MyControl),
    multi: true,
  }],
})
export class MyControl implements ControlValueAccessor {
  private _onChange = (_: any) => {};
  private _onTouched = () => {};
  writeValue(v: any) { ... }
  registerOnChange(fn: any) { this._onChange = fn; }
  registerOnTouched(fn: any) { this._onTouched = fn; }
  setDisabledState(d: boolean) { ... }
}

// ✅ NEW — FormValueControl (3 lines!)
export class MyControl implements FormValueControl<T> {
  value = model<T>(initialValue);
}
